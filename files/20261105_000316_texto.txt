# Linux $
cd ~/certs
awk '/BEGIN CERTIFICATE/{n++} n>0{print > ("gdbundle-" n ".pem")}' gd_bundle_dv-r1-g2.crt.pem
cp c9d0957aee0ae15a.crt bancobase2026com-fullchain.crt
for f in gdbundle-*.pem; do
  s=$(openssl x509 -in "$f" -noout -subject | sed 's/^subject=//')
  i=$(openssl x509 -in "$f" -noout -issuer  | sed 's/^issuer=//')
  if [ "$s" != "$i" ]; then echo "Incluyo: $s"; { echo; cat "$f"; } >> bancobase2026com-fullchain.crt
  else echo "Omito root: $s"; fi
done
grep -c 'BEGIN CERTIFICATE' bancobase2026com-fullchain.crt
openssl verify -partial_chain -untrusted bancobase2026com-fullchain.crt c9d0957aee0ae15a.crt
