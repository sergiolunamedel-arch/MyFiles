<Rule name="BancoBase-Limpieza-Huerfanas" type="Standard">
  <Source><![CDATA[
import sailpoint.object.*;
import sailpoint.api.*;
import sailpoint.tools.*;
import java.util.*;

boolean DRY_RUN = true;     // true = simulacro. Cambiar a false para borrar de verdad
String SOLO = "aandrade";   // prueba con un solo usuario. Dejar "" para TODAS las huérfanas

Set protegidos = new HashSet(Arrays.asList("spadmin", "AttributeSyncWorkGroup", "Everyone"));
List candidatos = new ArrayList();
int revisadas = 0, saltadas = 0, borradas = 0;

Iterator it = context.search(Identity.class, new QueryOptions(), "id, name");
List pares = new ArrayList();
while (it.hasNext()) { pares.add((Object[]) it.next()); }
Util.flushIterator(it);

for (Object[] r : pares) {
  String id = (String) r[0];
  String nombre = (String) r[1];
  revisadas++;
  if (nombre == null || protegidos.contains(nombre)) { saltadas++; continue; }
  boolean estiloAD = nombre.startsWith("DOMBASE\\");
  boolean estiloCorreo = nombre.indexOf("@") > 0 && nombre.indexOf("\\") < 0;
  if (!estiloAD && !estiloCorreo) { continue; }
  if (SOLO.length() > 0 && nombre.toLowerCase().indexOf(SOLO.toLowerCase()) < 0) { continue; }

  Identity i = context.getObjectById(Identity.class, id);
  if (i == null) continue;

  boolean tieneIntelexion = false;
  List links = i.getLinks();
  if (links != null) {
    for (Object o : links) {
      Link l = (Link) o;
      String app = l.getApplicationName();
      if (app != null && app.toLowerCase().indexOf("intelexion") >= 0) { tieneIntelexion = true; }
    }
  }
  boolean sinCaps = (i.getCapabilities() == null || i.getCapabilities().isEmpty());

  if (!tieneIntelexion && sinCaps && !i.isWorkgroup()) {
    int nLinks = (links == null) ? 0 : links.size();
    candidatos.add(nombre + "  (cuentas: " + nLinks + ")");
    if (!DRY_RUN) {
      new Terminator(context).deleteObject(i);
      context.commitTransaction();
      borradas++;
    }
  } else { saltadas++; }
  context.decache();
}

StringBuilder sb = new StringBuilder();
sb.append("MODO: ").append(DRY_RUN ? "SIMULACRO (no borra)" : "BORRADO REAL").append("\n");
sb.append("Filtro: ").append(SOLO.length() > 0 ? SOLO : "(todas)").append("\n");
sb.append("Revisadas: ").append(revisadas).append(" | Candidatas: ").append(candidatos.size())
  .append(" | Borradas: ").append(borradas).append(" | Saltadas: ").append(saltadas).append("\n");
int n = 0;
for (Object c : candidatos) { if (n++ < 20) sb.append("  ").append(c).append("\n"); }
return sb.toString();
  ]]></Source>
