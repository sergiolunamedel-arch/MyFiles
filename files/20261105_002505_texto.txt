Paso A — Respaldar y ver la configuración actual de Apache
bash
# Linux $
sudo ss -ltnp | grep -E ':8008\b'          # debe salir vacío
sudo cp -a /etc/httpd/conf/httpd.conf /var/tmp/httpd.conf.antes-nginx-$(date +%F)
sudo cp -a /etc/httpd/conf.d /var/tmp/httpd-conf.d.antes-nginx-$(date +%F)
sudo grep -rn -E '^\s*(Listen|<VirtualHost)' /etc/httpd/conf /etc/httpd/conf.d

Lo normal es que salga solo Listen 80 en httpd.conf y, si existen, uno o varios <VirtualHost *:80>. Si aparece un Listen 443 u otro puerto, detente y pásame la salida.

Paso B — Mover Apache a 127.0.0.1:8008
bash
# Linux $
sudo sed -i -E 's/^\s*Listen\s+(0\.0\.0\.0:)?80\s*$/Listen 127.0.0.1:8008/' /etc/httpd/conf/httpd.conf
sudo sed -i -E 's/<VirtualHost (\*|_default_|0\.0\.0\.0):80>/<VirtualHost *:8008>/' /etc/httpd/conf/httpd.conf /etc/httpd/conf.d/*.conf
sudo grep -rn -E '^\s*(Listen|<VirtualHost)' /etc/httpd/conf /etc/httpd/conf.d
sudo apachectl configtest                  # debe decir "Syntax OK"
sudo systemctl restart httpd
sudo ss -ltnp | grep -E ':(80|8008)\b'     # httpd solo en 127.0.0.1:8008, nada en el 80
curl -sI http://127.0.0.1:8008/            # debe responder 200
Paso C — Apuntar nginx al 8008 y arrancarlo
bash
# Linux $
sudo sed -i 's#proxy_pass http://127.0.0.1:8080;#proxy_pass http://127.0.0.1:8008;#' /etc/nginx/conf.d/scan.bancobase.com.conf
sudo grep -n 'proxy_pass\|listen' /etc/nginx/conf.d/scan.bancobase.com.conf
sudo nginx -t
sudo systemctl restart nginx
sudo systemctl status nginx --no-pager
Paso D — Probar
bash
# Linux $
curl -sI --resolve scan.bancobase.com:443:127.0.0.1 https://scan.bancobase.com/
curl -sI --resolve scan.bancobase.com:80:127.0.0.1  http://scan.bancobase.com/
