Pega estas consultas, una por una. Cada una termina en ;:
sql
SELECT COUNT(*) AS total,
       SUM(email IS NULL) AS sin_email,
       SUM(manager IS NULL) AS sin_manager
FROM spt_identity;
sql
SELECT COUNT(*) AS identidades_sin_cuentas
FROM spt_identity i
WHERE NOT EXISTS (SELECT 1 FROM spt_link l WHERE l.identity_id = i.id);
sql
SELECT a.name AS aplicacion, COUNT(*) AS cuentas
FROM spt_link l JOIN spt_application a ON l.application = a.id
GROUP BY a.name;
