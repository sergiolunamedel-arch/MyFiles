<Rule name="BancoBase-Limpieza-Huerfanas" type="Standard">
  <Source><![CDATA[
import sailpoint.object.*;
import sailpoint.api.*;
import sailpoint.tools.*;
import java.util.*;

boolean DRY_RUN = true;   // <-- cambiar a false SOLO después de revisar el simulacro

Set protegidos = new HashSet(Arrays.asList("spadmin", "AttributeSyncWorkGroup", "Everyone"));
List candidatos = new ArrayList();
int revisadas = 0, saltadas = 0;

Iterator it = context.search(Identity.class, new QueryOptions(), "id, name");
List pares = new ArrayList();
while (it.hasNext()) { Object[] r = (Object[]) it.next(); pares.add(r); }
Util.flushIterator(it);

for (Object[] r : pares) {
  String id = (String) r[0];
  String nombre = (String) r[1];
  revisadas++;
  if (nombre == null || protegidos.contains(nombre)) { saltadas++; continue; }
  boolean estiloAD = nombre.startsWith("DOMBASE\\");
  boolean estiloCorreo = nombre.indexOf("@") > 0 && nombre.indexOf("\\") < 0;
  if (!estiloAD && !estiloCorreo) { continue; }

  Identity i = context.getObjectById(Identity.class, id);
  if (i == null) continue;
  boolean sinLinks = (i.getLinks() == null || i.getLinks().isEmpty());
  boolean sinCaps = (i.getCapabilities() == null || i.getCapabilities().isEmpty());
  if (sinLinks && sinCaps && !i.isWorkgroup()) {
    candidatos.add(nombre);
    if (!DRY_RUN) {
      new Terminator(context).deleteObject(i);
    }
  } else { saltadas++; }
  context.decache(i);
}

StringBuilder sb = new StringBuilder();
sb.append("MODO: ").append(DRY_RUN ? "SIMULACRO (no borra)" : "BORRADO REAL").append("\n");
sb.append("Revisadas: ").append(revisadas).append(" | Candidatas: ").append(candidatos.size())
  .append(" | Saltadas: ").append(saltadas).append("\n");
int n = 0;
for (Object c : candidatos) { if (n++ < 15) sb.append("  ejemplo: ").append(c).append("\n"); }
return sb.toString();
  ]]></Source>
