Mándame esa salida. Mientras tanto, ya podemos ir directo al fix del JAVA_HOME explícito, ya que todo apunta a lo mismo:

bash
cd /opt/cloudgateway/apache-tomcat-9.0.104/bin
sudo -u tomcat ./shutdown.sh 2>/dev/null
sudo -u tomcat JAVA_HOME=$(dirname $(dirname $(readlink -f $(which java)))) ./startup.sh
sleep 3
ps -ef | grep -i cloudgateway
sudo tail -20 /opt/cloudgateway/apache-tomcat-9.0.104/logs/catalina.out
