SELECT name, extended1, extended2 FROM spt_identity
WHERE name = 'DOMBASE\\aandrade' OR extended2 LIKE 'aandrade@%' OR name LIKE 'aandrade@%';

SELECT COUNT(*) AS total_ad,
       SUM(l.attributes LIKE '%key="employeeID"%') AS con_employeeid
FROM spt_link l JOIN spt_application a ON a.id = l.application
WHERE a.name = 'Active Directory - QA';
