SELECT COUNT(*) AS identidades_sin_cuentas
FROM spt_identity i
WHERE NOT EXISTS (SELECT 1 FROM spt_link l WHERE l.identity_id = i.id);

SELECT COUNT(*) AS huerfanas_AD
FROM spt_identity WHERE name LIKE 'DOMBASE%';
sql
SELECT cuentas, COUNT(*) AS identidades
FROM (SELECT identity_id, COUNT(*) AS cuentas FROM spt_link GROUP BY identity_id) t
GROUP BY cuentas ORDER BY cuentas;

SELECT COUNT(*) AS total,
       SUM(email IS NOT NULL) AS con_email,
       SUM(manager IS NOT NULL) AS con_manager
FROM spt_identity;
