# 2. ¿Qué DocumentRoot y qué VirtualHosts usa Apache?
sudo apachectl -S 2>&1 | head -20
sudo grep -rn -E '^\s*(DocumentRoot|DirectoryIndex|Alias)' /etc/httpd/conf /etc/httpd/conf.d

# 3. ¿Dónde quedó tu index.html nuevo y qué hay en el DocumentRoot?
sudo find / -xdev -name 'index.htm*' -mmin -120 2>/dev/null
sudo ls -la /var/www/html/
