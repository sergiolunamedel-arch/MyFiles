SELECT a.name AS aplicacion,
       SUM(i.name LIKE 'DOMBASE%' OR i.name LIKE '%@%') AS en_identidad_huerfana,
       SUM(NOT (i.name LIKE 'DOMBASE%' OR i.name LIKE '%@%')) AS en_identidad_buena
FROM spt_link l
JOIN spt_application a ON a.id = l.application
JOIN spt_identity i ON i.id = l.identity_id
GROUP BY a.name;

SELECT SUM(name LIKE 'DOMBASE%') AS vacias_AD,
       SUM(name LIKE '%@%') AS vacias_Google,
       SUM(name NOT LIKE 'DOMBASE%' AND name NOT LIKE '%@%') AS otras
FROM spt_identity i
WHERE NOT EXISTS (SELECT 1 FROM spt_link l WHERE l.identity_id = i.id);

SELECT name FROM spt_identity i
WHERE NOT EXISTS (SELECT 1 FROM spt_link l WHERE l.identity_id = i.id)
  AND name NOT LIKE 'DOMBASE%' AND name NOT LIKE '%@%'
LIMIT 50;
